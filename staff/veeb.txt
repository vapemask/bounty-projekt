@app.route('/')
def index():
    cpu = psutil.cpu_percent()
    ram = psutil.virtual_memory().percent
    logid = ""
    if os.path.exists("raha_masin.log"):
        with open("raha_masin.log", "r") as f:
            logid = "".join(f.readlines()[-50:])
    targets = ""
    if os.path.exists("targets.txt"):
        with open("targets.txt", "r") as f:
            targets = f.read()
    return render_template_string(HTML_TEMPLATE, cpu=cpu, ram=ram, logid=logid, targets=targets)

@app.route('/add', methods=['POST'])
def add():
    d = request.form.get('d')
    if d:
        with open("targets.txt", "a") as f:
            f.write(f"{d}\n")
    return redirect('/')

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=8080)

import os, subprocess, time

def runda():
    while True:
        if os.path.exists("targets.txt"):
            with open("targets.txt", "r") as f:
                targets = [line.strip() for line in f if line.strip()]
            
            for domain in targets:
                print(f"--- RÜNDAN: {domain} ---")
                # 1. Leia alamdomeenid
                subprocess.run(f"subfinder -d {domain} -o subs.txt", shell=True)
                # 2. Kontrolli elusust

                # 3. Nuclei skaneerimine

                    cmd = "nuclei -l elusad.txt -severity critical,high -o leiud.txt"
